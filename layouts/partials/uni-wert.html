{{- /* Eine Angabe einer Uni als fertiges HTML (oder "" wenn leer), passend zur "art" der Spalte in data/uniguide-spalten.yaml. Für Tabelle, Vergleich und Uni-Seite gleich, damit eine neue Spalte überall gleich aussieht. Aufruf: partial "uni-wert.html" (dict "uni" $uni "spalte" $spalte) | safeHTML */ -}} {{- $s := .spalte -}} {{- $w := partial "uni-text.html" (index .uni $s.feld) -}} {{- $raus := "" -}} {{- if $w -}} {{- $art := $s.art | default "text" -}} {{- if eq $art "name" -}} {{- $raus = printf "%s" (printf "ems/uniguide/%s/" .uni.slug | relLangURL) (htmlEscape (partial "uni-name.html" .uni)) -}} {{- else if eq $art "ems" -}} {{- $raus = printf "%s" (htmlEscape $w) (i18n (printf "ems_status_%s" $w)) -}} {{- else if eq $art "sprachen" -}} {{- $raus = htmlEscape (partial "uni-sprachen.html" $w) -}} {{- else if eq $art "link" -}} {{- $raus = printf "%s" (htmlEscape $w) (htmlEscape $w) -}} {{- else if and (eq $art "punkte") (reflect.IsSlice $w) -}} {{- $li := "" -}} {{- range $w }}{{ $li = printf "%s
  • %s
    • " $li (htmlEscape .) }}{{ end -}} {{- $raus = printf "
      %s
    " $li -}} {{- else if reflect.IsSlice $w -}} {{- $raus = htmlEscape (delimit $w ", ") -}} {{- else -}} {{- /* Daten («15.02.2027», Stand «2026-09-30») ausgeschrieben in der Sprache der Seite: «15. Februar 2027», «15 février 2027» */ -}} {{- $t := string $w -}} {{- $teile := findRESubmatch `^(\d{2})\.(\d{2})\.(\d{4})$` $t 1 -}} {{- if $teile -}} {{- $d := index $teile 0 -}} {{- $t = time.Format ":date_long" (time.AsTime (printf "%s-%s-%s" (index $d 3) (index $d 2) (index $d 1))) -}} {{- else if findRE `^\d{4}-\d{2}-\d{2}$` $t -}} {{- $t = time.Format ":date_long" (time.AsTime $t) -}} {{- end -}} {{- $raus = htmlEscape $t -}} {{- end -}} {{- end -}} {{- return $raus -}}
